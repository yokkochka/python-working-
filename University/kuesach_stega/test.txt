def embed_text(wav_in, wav_out, text_file):
    # читаем текст
    text = read_text(text_file)
    bits = text_to_bits(text)
    bit_len = len(bits)

    print(f"Сообщение: {text}")
    print(f"Битов: {bit_len}")

    # открываем wav
    with wave.open(wav_in, 'rb') as wav:
        params = wav.getparams()
        frames = bytearray(wav.readframes(wav.getnframes()))

    # проверка вместимости
    if bit_len > len(frames):
        raise ValueError("Файл слишком маленький!")

    # встраивание
    for i in range(bit_len):
        frames[i] = (frames[i] & 254) | int(bits[i])

    # сохраняем
    with wave.open(wav_out, 'wb') as out:
        out.setparams(params)
        out.writeframes(frames)

    print("Встраивание завершено!")

def extract_text(wav_file, length):
    with wave.open(wav_file, 'rb') as wav:
        frames = bytearray(wav.readframes(wav.getnframes()))

    bits = []
    for i in range(length * 8):
        bits.append(str(frames[i] & 1))

    text = bits_to_text(''.join(bits))
    return text
